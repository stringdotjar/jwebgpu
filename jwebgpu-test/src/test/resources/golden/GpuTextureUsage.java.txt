package io.github.stringdotjar.jwebgpu;

/**
 * The bit flags of texture usage.
 *
 * <p>Combine flags with the bitwise or operator, for example {@code A | B}. The values are 64-bit numbers, so a
 * combination is a {@code long}.
 *
 * <p>C type: {@code WGPUTextureUsage}.
 */
public final class GpuTextureUsage {

  /** The none flag. */
  public static final long NONE = 0x0L;

  /** The copy src flag. */
  public static final long COPY_SRC = 0x1L;

  /** The copy dst flag. */
  public static final long COPY_DST = 0x2L;

  /** The texture binding flag. */
  public static final long TEXTURE_BINDING = 0x4L;

  /** The storage binding flag. */
  public static final long STORAGE_BINDING = 0x8L;

  /** The render attachment flag. */
  public static final long RENDER_ATTACHMENT = 0x10L;

  /** The transient attachment flag. */
  public static final long TRANSIENT_ATTACHMENT = 0x20L;

  private GpuTextureUsage() {}
}
