package io.github.stringdotjar.jwebgpu;

import org.jetbrains.annotations.Nullable;

/**
 * The contract between the API and one platform backend.
 *
 * <p>A backend (desktop, Android or web) implements this interface once. It carries the functions of the WebGPU API
 * that do not belong to any object, which is where everything starts: the backend creates the first
 * {@link GpuInstance}, and every other object is created from it, so each object is implemented by the same backend
 * that made it.
 *
 * <p>Programs do not call a backend directly. A platform installs its backend once with
 * {@link JWebGpu#install(GpuBackend)}, and {@link JWebGpu} forwards the global functions to it.
 */
public interface GpuBackend {

  /**
   * Returns a short name for this backend, for logs and error messages.
   *
   * @return The name, such as {@code ffm}, {@code jni} or {@code teavm}.
   */
  String name();

  /**
   * Create a WGPUInstance.
   *
   * @param descriptor The descriptor.
   * @return The gpu instance.
   */
  GpuInstance createInstance(@Nullable GpuInstanceDescriptor descriptor);

  /**
   * Get the list of {@code GpuInstanceFeatureName} values supported by the instance.
   *
   * @param features The features. Pass an object to fill. The call overwrites its fields and copies every string, so
   *     there is nothing to free afterwards.
   */
  void getInstanceFeatures(GpuSupportedInstanceFeatures features);

  /**
   * Get the limits supported by the instance.
   *
   * @param limits The limits. Pass an object to fill. The call overwrites its fields and copies every string, so there
   *     is nothing to free afterwards.
   * @return Indicates if there was an {@code OutStructChainError}.
   */
  @Nullable
  GpuStatus getInstanceLimits(GpuInstanceLimits limits);

  /**
   * Check whether a particular {@code GpuInstanceFeatureName} is supported by the instance.
   *
   * @param feature The feature.
   * @return {@code true} or {@code false}, depending on the check.
   */
  boolean hasInstanceFeature(GpuInstanceFeatureName feature);

  /**
   * Returns the procedure address (function pointer) of the named function. Cast the result to the proc pointer type of
   * that function.
   *
   * @param procName The proc name.
   * @return The proc address.
   */
  long getProcAddress(String procName);
}
