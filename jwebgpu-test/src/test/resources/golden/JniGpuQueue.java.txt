package io.github.stringdotjar.jwebgpu.jni;

import io.github.stringdotjar.jwebgpu.GpuBuffer;

/**
 * The Android implementation of {@link GpuQueue}, built on Dawn's {@code WGPUQueue}.
 *
 * <p>The wrapper stores only the native handle, as a {@code long}. Calls are marshalled into thread-local scratch
 * memory, so they allocate nothing. Methods that create objects also come in a refilling variant that reuses an
 * existing wrapper, so per-frame objects need no Java allocation either.
 *
 * <p>This class is generated from the specification.
 */
public final class JniGpuQueue extends GpuQueue {

  private long address;

  /**
   * Creates an empty wrapper that holds no native object.
   *
   * <p>It reports {@link #isReleased()} until a refilling method stores a handle in it.
   */
  public JniGpuQueue() {}

  /**
   * Wraps a native handle. The wrapper owns the reference, so release it when done.
   *
   * @param address The native {@code WGPUQueue} handle. It must not be 0.
   */
  public JniGpuQueue(long address) {
    this.address = address;
  }

  @Override
  public long onSubmittedWorkDone(GpuCallbackMode mode, GpuQueueWorkDoneCallback callback) {
    long self = live();
    Scratch s = Scratch.begin();
    if (callback == null) {
      throw new NullPointerException("callback");
    }
    long cb = s.alloc(40L);
    s.putLong(cb + 0L, 0L);
    s.putInt(cb + 8L, mode.value);
    s.putLong(cb + 16L, JniCallbacks.QUEUE_WORK_DONE);
    s.putLong(cb + 24L, CallbackTable.put(callback));
    s.putLong(cb + 32L, 0L);
    return Jni.wgpuQueueOnSubmittedWorkDone(self, cb);
  }

  @Override
  public void setLabel(String label) {
    long self = live();
    Scratch s = Scratch.begin();
    Jni.wgpuQueueSetLabel(self, s.view(label));
  }

  @Override
  public void submit(GpuCommandBuffer[] commands, int commandCount) {
    long self = live();
    Scratch s = Scratch.begin();
    long n0 = 0L;
    if (commandCount > 0) {
      n0 = s.alloc(commandCount * 8L);
      for (int i = 0; i < commandCount; i++) {
        s.putLong(n0 + i * 8L, JniGpuCommandBuffer.addr(commands[i]));
      }
    }
    Jni.wgpuQueueSubmit(self, commandCount, n0);
  }

  @Override
  public void writeBuffer(GpuBuffer buffer, long bufferOffset, ByteBuffer data) {

    Scratch s = Scratch.begin();
    long n0 = s.input(data);
    Jni.wgpuQueueWriteBuffer(self, JniGpuBuffer.addr(buffer), bufferOffset, n0, data.remaining());
  }

  }

  @Override
  public void release() {
    long a = address;
    if (a != 0L) {
      address = 0L;
      Jni.wgpuQueueRelease(a);
    }
  }

    Jni.wgpuQueueAddRef(live());
  }

  @Override

    return address == 0L;
  }

   * @return The native handle.

   * @throws IllegalStateException If the object was released.
   */

    if (o instanceof JniGpuQueue f) {
      long a = f.address;
      if (a == 0L) {
        throw new IllegalStateException("The GpuQueue was released");
      }
      return a;
    }

      throw new NullPointerException("queue");
    }
    throw new IllegalArgumentException("The GpuQueue was not created by the JNI backend");
  }

    address = handle;
    if (old != 0L) {
      Jni.wgpuQueueRelease(old);
    }
  }

  private long live() {
    long a = address;
    if (a == 0L) {
      throw new IllegalStateException("The GpuQueue was released");
    }
    return a;
  }

   * Returns the native handle.

  }
