package io.github.stringdotjar.jwebgpu.jni;

/**
 * The native functions of the JNI backend, one per C function of {@code webgpu.h}.
 *
 * <p>The C glue library registers all of them at once when it loads. A parameter is a {@code long} for every handle and
 * pointer, and a struct that C takes by value (a string view or a callback info) is passed as the address of the struct
 * in scratch memory. This class is generated from the specification.
 */

  private Jni() {}

  /**
   * Returns the address of the C function that receives one kind of callback.
   *
   * @param index The position of the callback in the specification.
   * @return The function address.
   */
  static native long callbackAddress(int index);

  static native long wgpuCreateInstance(long a0);
  static native void wgpuGetInstanceFeatures(long a0);
  static native int wgpuGetInstanceLimits(long a0);
  static native int wgpuHasInstanceFeature(int a0);
  static native long wgpuGetProcAddress(long a0);
  static native void wgpuAdapterGetFeatures(long a0, long a1);
  static native int wgpuAdapterGetInfo(long a0, long a1);
  static native int wgpuAdapterGetLimits(long a0, long a1);
  static native int wgpuAdapterHasFeature(long a0, int a1);
  static native long wgpuAdapterRequestDevice(long a0, long a1, long a2);
  static native void wgpuAdapterAddRef(long a0);
  static native void wgpuAdapterRelease(long a0);
  static native void wgpuAdapterInfoFreeMembers(long a0);
  static native void wgpuBindGroupSetLabel(long a0, long a1);
  static native void wgpuBindGroupAddRef(long a0);
  static native void wgpuBindGroupRelease(long a0);
  static native void wgpuBindGroupLayoutSetLabel(long a0, long a1);
