package io.github.stringdotjar.jwebgpu.teavm;

import org.teavm.jso.JSMethod;
import org.teavm.jso.JSObject;
import org.teavm.jso.JSProperty;
import org.teavm.jso.core.JSPromise;

/**
 * The JavaScript object interface of the browser's {@code GPUBuffer}.
 *
 * <p>The methods call the browser directly, so they allocate no Java objects. Most programs use the wrapper classes of
 * the API instead; this interface is for code that needs a feature the shared API does not have.
 *
 * <p>This interface is generated from the WebGPU WebIDL.
 */
public interface JsGpuBuffer extends JSObject {

  /**
   * Reads the {@code size} attribute.
   *
   * @return The value of the attribute.
   */
  @JSProperty("size")
  double getSize();

  /**
   * Reads the {@code usage} attribute.
   *
   * @return The value of the attribute.
   */
  @JSProperty("usage")
  int getUsage();

  /**
   * Reads the {@code mapState} attribute.
   *
   * @return The value of the attribute.
   */
  @JSProperty("mapState")
  String getMapState();

  /**
   * Reads the {@code label} attribute.
   *
   * @return The value of the attribute.
   */
  @JSProperty("label")
  String getLabel();

  /**
   * Writes the {@code label} attribute.
   *
   * @param value The new value.
   */
  @JSProperty("label")
  void setLabel(String value);

  /**
   * Calls {@code mapAsync} of the browser object. The last 2 optional argument(s) are left out.
   *
   * @param mode The {@code mode} argument.
   * @return The result of the call.
   */
  @JSMethod("mapAsync")
  JSPromise<JSObject> mapAsync(int mode);

  /**
   * Calls {@code mapAsync} of the browser object. The last 1 optional argument(s) are left out.
   *
   * @param mode The {@code mode} argument.
   * @param offset The {@code offset} argument.
   * @return The result of the call.
   */
  @JSMethod("mapAsync")
  JSPromise<JSObject> mapAsync(int mode, double offset);

  /**
   * Calls {@code mapAsync} of the browser object.
   *
   * @param mode The {@code mode} argument.
   * @param offset The {@code offset} argument.
   * @param size The {@code size} argument.
   * @return The result of the call.
   */
  @JSMethod("mapAsync")
  JSPromise<JSObject> mapAsync(int mode, double offset, double size);

  /**
   * Calls {@code getMappedRange} of the browser object. The last 2 optional argument(s) are left out.
   *
   * @return The result of the call.
   */
  @JSMethod("getMappedRange")
  JSObject getMappedRange();

  /**
   * Calls {@code getMappedRange} of the browser object. The last 1 optional argument(s) are left out.
   *
   * @param offset The {@code offset} argument.
   * @return The result of the call.
   */
  @JSMethod("getMappedRange")
  JSObject getMappedRange(double offset);

  /**
   * Calls {@code getMappedRange} of the browser object.
   *
   * @param offset The {@code offset} argument.
   * @param size The {@code size} argument.
   * @return The result of the call.
   */
  @JSMethod("getMappedRange")
  JSObject getMappedRange(double offset, double size);

  /** Calls {@code unmap} of the browser object. */
  @JSMethod("unmap")
  void unmap();

  /** Calls {@code destroy} of the browser object. */
  @JSMethod("destroy")
  void destroy();
}
