package io.github.stringdotjar.jwebgpu;

import org.jetbrains.annotations.Nullable;

/**
 * Receives the result of an operation that finishes later.
 *
 * <p>The native userdata pointers of the C callback do not exist in Java: capture whatever you need in a lambda or a
 * field instead. Create the callback once and reuse it, so that nothing is allocated when it is passed to the API.
 *
 * <p>C type: {@code WGPURequestAdapterCallback}.
 */
@FunctionalInterface
public interface GpuRequestAdapterCallback {

  /**
   * Called when the operation has finished.
   *
   * @param status The status. It is {@code null} if the native library reports a value this API does not know.
   * @param adapter The adapter, or {@code null} if the operation failed.
   * @param message The message. It is {@code null} if the native library passes no text.
   */
  void onRequestAdapter(
      @Nullable GpuRequestAdapterStatus status,
      @Nullable GpuAdapter adapter,
      @Nullable String message);
}
