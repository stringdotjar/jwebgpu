package io.github.stringdotjar.jwebgpu;

import org.jetbrains.annotations.Nullable;

/**
 * A reusable render pass color attachment struct.
 *
 * <p>Create it once and refill the fields, because a new descriptor allocates. {@link #reset()} restores the defaults
 * from the C header.
 *
 * <p>C type: {@code WGPURenderPassColorAttachment}.
 */
public final class GpuRenderPassColorAttachment {

  /**
   * The first extension struct chained onto this one, or {@code null} for none.
   *
   * <p>Extension structs form a singly linked list through their {@code next} field.
   */
  public @Nullable GpuChainedStruct nextInChain;

  /**
   * If {@code NULL}, indicates a hole in the parent
   * {@code GpuRenderPassDescriptor::colorAttachments} array.
   */
  public @Nullable GpuTextureView view;

  /** The depth slice. */
  public int depthSlice;

  /** The resolve target. */
  public @Nullable GpuTextureView resolveTarget;

  /** The load op. */
  public GpuLoadOp loadOp;

  /** The store op. */
  public GpuStoreOp storeOp;

  /** The clear value. */
  public final GpuColor clearValue = new GpuColor();

  /** Creates the struct with the defaults of the C header. */
  public GpuRenderPassColorAttachment() {
    reset();
  }

  /**
   * Restores every field to the default of the C header.
   *
   * <p>Nested structs and array elements are reset too. Arrays keep their capacity, and nothing is allocated.
   */
  public void reset() {
    nextInChain = null;
    view = null;
    depthSlice = GpuConstants.DEPTH_SLICE_UNDEFINED;
    resolveTarget = null;
    loadOp = GpuLoadOp.UNDEFINED;
    storeOp = GpuStoreOp.UNDEFINED;
    clearValue.reset();
  }
}
