package io.github.stringdotjar.jwebgpu;

/**
 * Chained in {@code GpuSurfaceDescriptor} to make an {@code GpuSurface} wrapping an
 * <a href="https://www.x.org/releases/current/doc/libX11/libX11/libX11.html">Xlib</a> {@code Window}.
 *
 * <p>This is an extension struct. Attach it to the {@code nextInChain} field of {@link GpuSurfaceDescriptor}.
 *
 * <p>Create it once and refill the fields, because a new descriptor allocates. {@link #reset()} restores the defaults
 * from the C header.
 *
 * <p>C type: {@code WGPUSurfaceSourceXlibWindow}.
 */
public final class GpuSurfaceSourceXlibWindow extends GpuChainedStruct {

  /**
   * A pointer to the
   * <a href="https://www.x.org/releases/current/doc/libX11/libX11/libX11.html#Opening_the_Display">{@code Display}</a>
   * connected to the X server.
   */
  public long display;

  /**
   * The <a href="https://www.x.org/releases/current/doc/libX11/libX11/libX11.html#Creating_Windows">{@code Window}</a>
   * that will be wrapped by the {@code GpuSurface}.
   */
  public long window;

  /** Creates the struct with the defaults of the C header. */
  public GpuSurfaceSourceXlibWindow() {
    reset();
  }

  /** Restores every field to the default of the C header. */
  public void reset() {
    next = null;
    display = 0L;
    window = 0L;
  }

  /**
   * Returns the tag that tells the native library which extension struct this is.
   *
   * @return The native {@code WGPUSType} number of this struct.
   */
  @Override
  public int sType() {
    return GpuSType.SURFACE_SOURCE_XLIB_WINDOW.value;
  }
}
